```python
# --- 简易 Embedding 模拟（字符级 TF-IDF 风格）---
def tokenize(text):
    """简易分词：按空格+标点分割，取 2-gram"""
    import re
    tokens = re.findall(r'\w+', text.lower())
    bigrams = [f"{tokens[i]}_{tokens[i+1]}" for i in range(len(tokens)-1)]
    return tokens + bigrams

def cosine_similarity(vec1, vec2):
    """余弦相似度"""
    intersection = set(vec1.keys()) & set(vec2.keys())
    dot = sum(vec1[k] * vec2[k] for k in intersection)
    norm1 = math.sqrt(sum(v**2 for v in vec1.values()))
    norm2 = math.sqrt(sum(v**2 for v in vec2.values()))
    return dot / (norm1 * norm2) if norm1 and norm2 else 0

def retrieve(query, kb, top_k=3):
    """RAG 检索：计算 query 与每个文档的相似度，返回 top_k"""
    query_vec = Counter(tokenize(query))
    scored = []
    for doc in kb:
        doc_vec = Counter(tokenize(doc["text"]))
        sim = cosine_similarity(query_vec, doc_vec)
        scored.append((sim, doc))
    scored.sort(key=lambda x: x[0], reverse=True)
    return [doc for _, doc in scored[:top_k]]


# --- 实验 1：静态 Context（全量注入）---
def static_context_experiment(question, kb):
    """把所有知识塞入 context"""
    full_context = "\n\n".join([f"[{doc['id']}] {doc['text']}" for doc in kb])
    token_count = len(full_context.split())  # 简易 token 估算
    return {
        "mode": "static",
        "question": question,
        "context_chars": len(full_context),
        "context_tokens_est": token_count,
        "relevant_docs": "ALL",
        "efficiency": f"{token_count} tokens for 1 question",
    }


# --- 实验 2：RAG（动态检索）---
def rag_experiment(question, kb, top_k=3):
    """只注入检索到的相关文档"""
    relevant = retrieve(question, kb, top_k)
    context = "\n\n".join([f"[{doc['id']}] {doc['text']}" for doc in relevant])
    token_count = len(context.split())
    return {
        "mode": "rag",
        "question": question,
        "context_chars": len(context),
        "context_tokens_est": token_count,
        "relevant_docs": [doc["id"] for doc in relevant],
        "efficiency": f"{token_count} tokens (vs static ALL)",
    }
```

### 1.2 关键观察

| 维度 | 静态 Context | RAG |
|------|-------------|-----|
| Token 消耗 | O(N) — 线性增长 | O(K) — K = top_k，常数级 |
| 噪声比 | 高（不相关内容干扰模型注意力） | 低（只注入相关文档） |
| 延迟 | 一次 API 调用 | 检索 + API 调用 |
| 知识库上限 | ~128K token 窗口限制 | 理论上无上限 |
| 适合场景 | 知识库 <10 条 | 知识库 >20 条 |

```python
"""experiments/4_module1_connections/4_rag_to_agent.py
2026-05-25 — ReAct Agent 实战实现

核心架构：
- Agent 使用 LLM 驱动 Thought → Action → Observation 循环
- 工具集：get_calldata, is_verified_contract, get_contract_abi, simulate_tx
- 与 5/23 的 mock 版本的区别：LLM 做推理决策，而非硬编码 if-else

import json

# --- 配置 ---

client = OpenAI(
    api_key=DEEPSEEK_API_KEY,

# --- 工具集（与 5/23 mock 版本相比，新增 simulation）---
def get_calldata(tx_hash):

def is_verified_contract(address):

        "0x7a250d5630B4cF539739dF2C5dAcb4c659F2488D",  # Uniswap V2 Router
        "0xdef1c0ded9bec7f1a1670819833240f027b25eff",  # 0x Exchange Proxy
        "0x5FF137D4b0FDCD49DcA30c7CF57E578a026d2789",  # ERC-4337 EntryPoint

def simulate_tx(tx_data):
    """模拟交易执行结果"""
    selector = tx_data.get("selector", "")
    to_addr = tx_data.get("to", "")

    warnings = []
    if "transferFrom" in selector:
        warnings.append("⚠️ transferFrom 检测：确认 spender 已得到你的授权")
    if not is_verified_contract(to_addr)["verified"]:
        warnings.append("⚠️ 目标合约未验证：可能包含恶意逻辑")

    return {
        "will_revert": "transferFrom" in selector and not is_verified_contract(to_addr)["verified"],
        "estimated_gas": 150000,
        "warnings": warnings,
    }

# --- ReAct Agent 核心：LLM 驱动的推理-行动循环 ---
SYSTEM_PROMPT = """你是一个链上安全分析 Agent。你的任务是通过多步推理分析交易安全性。

你有以下工具可用：
- get_calldata(tx_hash): 获取交易 calldata
- is_verified_contract(address): 检查合约是否已验证
- get_contract_abi(address): 获取合约 ABI
- simulate_tx(tx_data): 模拟交易执行

你必须按照以下格式响应（严格遵循，每步只能调用一个工具）：

Thought: <你的推理>
Action: <工具名称>(<参数>)

或者当分析完成时：
Thought: <最终推理>
Final Answer: <简洁的安全结论，包含风险等级和原因>

风险等级：LOW（已验证合约的正常交互）/ MEDIUM（需要用户确认）/ HIGH（明确的危险信号）
"""


def react_agent_loop(tx_hash, max_steps=5):
    """
    ReAct Agent 主循环
    1. LLM 推理 → 决定调用哪个工具
    2. 执行工具 → 获取结果
    3. 将结果反馈给 LLM → 继续推理
    4. 直到 LLM 输出 Final Answer 或达到最大步数
    """
    messages = [{"role": "system", "content": SYSTEM_PROMPT}]
    steps = []

    # 初始输入
    messages.append({
        "role": "user",
        "content": f"请分析交易 {tx_hash} 的安全性。一步一步来。"
    })

    for step_num in range(max_steps):
        # 调用 LLM
        response = client.chat.completions.create(
            model="deepseek-chat",
            messages=messages,
            temperature=0.3,
        )
        llm_output = response.choices[0].message.content

        # 解析 LLM 输出
        if "Final Answer:" in llm_output:
            # Agent 认为分析完成
            thought = llm_output.split("Final Answer:")[0].replace("Thought:", "").strip()
            final_answer = llm_output.split("Final Answer:")[1].strip()
            steps.append({
                "step": step_num + 1,
                "type": "final",
                "thought": thought,
                "answer": final_answer,
            })
            return {"risk_analysis": final_answer, "steps": steps, "total_steps": step_num + 1}

        # 解析 Action
        if "Action:" in llm_output:
            thought_part = llm_output.split("Action:")[0].replace("Thought:", "").strip()
            action_part = llm_output.split("Action:")[1].strip()

            # 提取工具名和参数
            tool_name = action_part.split("(")[0].strip()
            args_str = action_part.split("(")[1].rstrip(")").strip()

            # 执行工具
            tool = TOOLS.get(tool_name)
            if tool:
                # 根据工具名构造参数
                if tool_name == "get_calldata":
                    result = tool["fn"](tx_hash)
                elif tool_name == "is_verified_contract":
                    # 从之前的 observation 中找地址，或直接用 args_str
                    result = tool["fn"](args_str.strip('"\''))
                elif tool_name == "get_contract_abi":
                    result = tool["fn"](args_str.strip('"\''))
                elif tool_name == "simulate_tx":
                    # 用上次 get_calldata 的结果
                    last_calldata = None
                    for s in steps:
                        if s.get("tool") == "get_calldata":
                            last_calldata = s["result"]
                    result = tool["fn"](last_calldata or {})
                else:
                    result = {"error": f"未知工具: {tool_name}"}

                steps.append({
                    "step": step_num + 1,
                    "type": "action",
                    "thought": thought_part,
                    "tool": tool_name,
                    "args": args_str,
                    "result": result,
                })

                # 将工具结果反馈给 LLM
                messages.append({"role": "assistant", "content": llm_output})
                messages.append({
                    "role": "user",
                    "content": f"Observation: {json.dumps(result, ensure_ascii=False)}"
                })
            else:
                messages.append({
                    "role": "user",
                    "content": f"错误：未找到工具 '{tool_name}'。可用工具：{list(TOOLS.keys())}"
                })

    return {"risk_analysis": "分析超时", "steps": steps, "total_steps": max_steps}


# --- 运行测试 ---
if __name__ == "__main__":
    print("=" * 60)
    print("ReAct Agent — 交易安全分析")
    print("=" * 60)

    test_cases = [
        ("0xBBB", "正常 Uniswap swap"),
        ("0xAAA", "可疑 transferFrom"),
        ("0xCCC", "未知转账"),
    ]

    for tx_hash, desc in test_cases:
        print(f"\n{'─' * 50}")
        print(f"测试: {desc} ({tx_hash})")

        try:
            result = react_agent_loop(tx_hash, max_steps=5)
            print(f"  总步数: {result['total_steps']}")
            print(f"  分析: {result['risk_analysis'][:200]}...")

            # 打印每步详情
            for step in result["steps"]:
                if step["type"] == "final":
                    print(f"  Step {step['step']}: ✅ 完成")
                else:
                    print(f"  Step {step['step']}: 🔧 {step['tool']} → {str(step['result'])[:100]}")

        except Exception as e:
            print(f"  ❌ 错误: {e}")

    print(f"\n{'=' * 60}")
    print("完成！对比 5/23 的硬编码版本：")
    print("  - Mock 版: if-else 决策，只能处理预设场景")
    print("  - LLM 版: LLM 做推理，可以处理未见过的交易模式")
    print("  - 下一步: 接入真实 Blockscout API + 真实合约验证")
```

### 2.2 ReAct Agent vs Mock Agent 对比

| 维度 | 5/23 Mock 版本 | 5/25 LLM 驱动版本 |
|------|---------------|-------------------|
| 决策逻辑 | 硬编码 if-else | LLM 推理 |
| 可扩展性 | 每种新模式需加代码 | LLM 自动泛化 |
| 工具调用 | 固定顺序 | LLM 动态决定 |
| 成本 | 零 API 调用 | 每步 1 次 API 调用 |
| 灵活性 | 低 | 高 |
| 适合场景 | 高度确定的任务 | 需要推理的开放场景 |

**核心洞察**：你的交易解释器（exp 2）更接近 Mock 版本——calldata 解析是硬编码的，LLM 只负责最终的「解释」步骤。升级路径：将 calldata 解析也交给 Agent 动态决定（用哪个 API、查什么字段），让 LLM 做全链路推理。

## 三、Module 2 预览：以 Agent 视角重审 Web3

### 3.1 Wallet — Agent 需要钱包的什么？

```
传统钱包视角                    Agent 钱包视角
─────────────                  ──────────────
用户 → 钱包 → 签名 → 发送       Agent → 钱包 → 自动签名 → 自动发送
                                         │
                              关键差异：
                              ① 没有人类在循环中确认每一步
                              ② 需要编程化的权限策略（不是「每次弹窗」）
                              ③ nonce 管理必须自动化
                              ④ gas 策略需要自适应
```

**Agent 钱包的核心问题**：

| 问题 | 传统方案 | Agent 适配 |
|------|---------|-----------|
| 签名触发 | 用户手动确认 | 编程策略（规则引擎 / AI 决策） |
| 私钥存储 | 用户本地 | HSM / MPC / TEE 安全环境 |
| 权限范围 | 全部资产 | 限额 + 白名单 + 时间窗口 |
| 交易队列 | 用户逐个签名 | 自动 nonce 管理 + 并行/串行 |
| 失败处理 | 用户重试 | 自动重试 + 降级策略 |

### 3.2 Smart Contract — Agent 与合约的交互模型

```
Agent 调用合约的标准流程：

1. 意图理解
   "用户想用 0.5 ETH 换 USDC"
          ↓
2. ABI 查询
   获取 Uniswap Router ABI
          ↓
3. calldata 构造
   swapExactETHForTokens(amountOutMin, path, to, deadline)
          ↓
4. 交易模拟 (eth_call / Tenderly)
   检查是否会 revert、gas 估算
          ↓
5. 签名 + 发送
   signTransaction → sendRawTransaction
          ↓
6. 确认
   等待交易上链 + 解析 event log
```

**你的交易解释器的位置**：目前卡在步骤 1→2（理解 calldata 但不会构造 calldata）。这恰好是 Module 2 和 Module 3 Bridge 之间的关键跃迁点。

### 💡 Insights & Reflections | 收获与思考

**What I Learned | 今日收获**:

1. **RAG 的「为什么不早用」时刻**：20 条文档的静态注入已消耗 ~2000 token，RAG 检索降到 ~300 token。当知识库从 20 条增长到 200 条时，RAG 不是「优化」而是「必需」。

2. **ReAct Agent 的本质是「把 if-else 外包给 LLM」**：对比 5/23 的 mock 版和今天的 LLM 版，同样的架构（Thought → Action → Observation），区别在于决策引擎从 `if selector == "transferFrom"` 变成了 LLM 的语义理解。这意味着 Agent 可以处理「你没有提前编码」的场景。

3. **Module 2 的 Agent 视角是关键**：传统 Wallet 章节讲「用户怎么用钱包」，但 Agent 需要的是「编程化的钱包能力」——不是每次弹窗确认，而是策略驱动的自动签名。

**Problems Encountered | 遇到的问题**:

1. ReAct 循环的 JSON 解析稳定性：LLM 输出的 Action 格式偶尔不规范（如括号不匹配）。解决方案：用 retry + 更严格的 System Prompt 约束输出格式。
2. DeepSeek API 调用延迟：每次 step ~1.5s，5 步约 7.5s。对实时风控场景偏慢，但对分析场景可接受。

**Thoughts | 思考与感悟**:

> Week 1 从「LLM 是什么」走到「让 LLM 驱动多步推理 Agent」，这个跃迁比我预期的快。关键在于：不要等学完所有章节再动手。在学 Prompt 时写 Prompt 实验，在学 RAG 时写检索代码，在学 Agent 时写 ReAct 循环——概念和代码交替推进。这个方法下周会延续到 Module 2：学 Wallet 时就让 Agent 去签名，学 Smart Contract 时就让 Agent 去构造 calldata。
